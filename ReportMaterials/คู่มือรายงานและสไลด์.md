# คู่มือเลือกภาพและโค้ดสำหรับรายงาน / สไลด์

เกมออกกำลังกายด้วยแผ่นเหยียบ 4 ทิศ (Exergame) ทำด้วย Unity 2022.3
มี 3 โหมดเกม: **ถ้ำกู้ระเบิด** (Tutorialmine, Cave1–3), **วิ่งไม่รู้จบ** (CookieRunner), **จำลูกศร** (Arrow Memory)

ภาพที่อยู่ในโฟลเดอร์ `images/` ได้จากไฟล์ภาพจริงในโปรเจกต์ ส่วนภาพตอนเล่นจริง (ข้อ 2) ต้องแคปจาก Unity เอง

## 3. โค้ดที่ควรใส่

### 3.1 ระบบรับปุ่มจากแผ่นเหยียบ — `Assets/Code/GameInputBindings.cs`
**ใช้อธิบาย:** แผ่นเหยียบต่อกับคอมเป็นจอยสติ๊ก แต่ละทิศคือปุ่มจอยหนึ่งปุ่ม รองรับคีย์บอร์ดสำรองด้วย และเปลี่ยนปุ่มได้จากหน้า Option

```csharp
// ★ ทิศของแผ่นเหยียบ 4 ทิศ
public const KeyCode PadUp    = KeyCode.Joystick1Button5;
public const KeyCode PadDown  = KeyCode.Joystick1Button11;
public const KeyCode PadLeft  = KeyCode.Joystick1Button8;
public const KeyCode PadRight = KeyCode.Joystick1Button3;

// แต่ละการกระทำ = ปุ่มแผ่นเหยียบ + ปุ่มคีย์บอร์ด + ปุ่มลูกศรสำรอง
{ GameAction.RunJump,  new Defaults(PadUp,   KeyCode.Space, KeyCode.UpArrow) },
{ GameAction.RunSlide, new Defaults(PadDown, KeyCode.S,     KeyCode.DownArrow) },
{ GameAction.MoveLeft, new Defaults(PadLeft, KeyCode.A,     KeyCode.LeftArrow) },

// ★ กดปุ่มไหนก็ได้ในชุดนี้ = ทำการกระทำนั้น
public static bool GetDown(GameAction action)
{
    EnsureLoaded();
    return Input.GetKeyDown(padBinding[action])
        || Input.GetKeyDown(keyBinding[action])
        || Input.GetKeyDown(defaults[action].fallback);
}
```

### 3.2 โหมดถ้ำ: นับระเบิดรอบช่อง + เปิดช่องว่างต่อเนื่อง — `Assets/Code/PlayerController.cs`
**ใช้อธิบาย:** หลักการแบบ Minesweeper ช่องที่เปิดแสดงจำนวนระเบิด/มอนสเตอร์ใน 8 ช่องรอบๆ ถ้าเป็น 0 จะเปิดช่องข้างเคียงต่อเองด้วย BFS (Queue)

```csharp
// ★ นับกับดักใน 8 ช่องรอบตัว
int CountBombsAround(Vector3Int targetCell)
{
    int count = 0;
    for (int x = -1; x <= 1; x++)
        for (int y = -1; y <= 1; y++)
        {
            if (x == 0 && y == 0) continue;
            Vector3Int c = new Vector3Int(targetCell.x + x, targetCell.y + y, targetCell.z);
            if (mineCells.Contains(c) || monsterCells.Contains(c)) count++;
        }
    return count;
}

// ★ เปิดช่องต่อเนื่องแบบ BFS เมื่อรอบๆ ไม่มีกับดัก
void RevealCascade(Vector3Int startCell)
{
    Queue<Vector3Int> cellsToProcess = new Queue<Vector3Int>();
    cellsToProcess.Enqueue(startCell);
    while (cellsToProcess.Count > 0)
    {
        Vector3Int currentCell = cellsToProcess.Dequeue();
        if (revealedCells.Contains(currentCell) || IsTileProtected(currentCell)) continue;
        RevealSingleCell(currentCell);
        if (CountBombsAround(currentCell) == 0)
            for (int x = -1; x <= 1; x++)
                for (int y = -1; y <= 1; y++)
                {
                    if (x == 0 && y == 0) continue;
                    Vector3Int n = new Vector3Int(currentCell.x + x, currentCell.y + y, currentCell.z);
                    if (!revealedCells.Contains(n)) cellsToProcess.Enqueue(n);
                }
    }
}
```
ส่วนเสริมที่เล่าได้: ปักธงถูกจุดเท่านั้นถึงนับว่าเคลียร์ (`ToggleFlagInFront`), Cave3 มีมอนสเตอร์เดินตามผู้เล่นบนตาราง (`MonsterChaserAI.cs`) และระเบิดทำดาเมจมอนสเตอร์ได้

### 3.3 โหมดวิ่ง: กระโดดแบบจำปุ่มล่วงหน้า (Jump Buffer) — `Assets/Code/CookieRun/CookiePlayerController.cs`
**ใช้อธิบาย:** ผู้เล่นเหยียบแผ่นอาจเร็วกว่าจังหวะลงพื้นเล็กน้อย เกมจะจำการเหยียบไว้ 0.3 วินาที พอเท้าแตะพื้นก็กระโดดให้ทันที ทำให้เล่นด้วยแผ่นเหยียบได้ลื่นไหล

```csharp
[SerializeField] private float jumpBufferTime = 0.3f;

private void HandleJumpBuffer()
{
    if (jumpBufferTimer > 0f)
    {
        jumpBufferTimer -= Time.deltaTime;
        // ★ ลงพื้นแล้ว (หรือเพิ่งเดินตกขอบไม่นาน) และยังอยู่ในช่วงจำปุ่ม = กระโดดเลย
        if (isGrounded || coyoteTimer > 0f)
        {
            PerformJump(false);
            jumpBufferTimer = 0f;
        }
    }
}
```

### 3.4 โหมดวิ่ง: ความยากตามระยะทาง + ฉากสุ่มสร้าง — `CookieGameManager.cs`, `CookieMapSpawner.cs`
**ใช้อธิบาย:** ยิ่งวิ่งไกล ระดับยิ่งสูง ความเร็วเพิ่มแต่จำกัดไม่เกิน 1.4 เท่า (เพราะแผ่นเหยียบไม่มีราวจับ) และฉากที่มีหลุมโผล่บ่อยขึ้น

```csharp
// ★ ระดับความยาก 8 ระดับ ตามระยะทาง (เมตร)
new DifficultyLevel { startDistance = 0f,    announcement = "" },
new DifficultyLevel { startDistance = 150f,  announcement = "ROLLING BOULDERS!" },
new DifficultyLevel { startDistance = 300f,  announcement = "FALLING STALACTITES!" },
new DifficultyLevel { startDistance = 450f,  announcement = "SPIKE TRAPS!" },
new DifficultyLevel { startDistance = 650f,  announcement = "MAX SPEED!" },
new DifficultyLevel { startDistance = 850f,  announcement = "GIANT MUSHROOMS!" },
new DifficultyLevel { startDistance = 1050f, announcement = "HANGING ROOTS!" },
new DifficultyLevel { startDistance = 1250f, announcement = "CRYSTAL SPIRES!" },
```

```csharp
// ★ สุ่มเลือกฉากแบบถ่วงน้ำหนัก: ระดับสูง ฉากหลุมมีโอกาสมากขึ้น
float pitWeight = 1f + pitWeightPerLevel * (level - 1);
float total = 0f;
foreach (GameObject c in spawnCandidates) total += ChunkWeight(c, pitWeight);
float genWeight = generated ? pitWeight * 2f : 0f;   // ฉากหลุมแบบสุ่มสร้าง

float r = Random.value * (total + genWeight);
foreach (GameObject c in spawnCandidates)
{
    r -= ChunkWeight(c, pitWeight);
    if (r <= 0f) return c;
}
return null;  // null = สร้างฉากหลุมแบบสุ่ม (SpawnGeneratedChunk)
```
ข้อกำหนดของฉากสุ่มสร้าง (ใส่เป็นตารางในรายงานได้): หลุมกว้าง 2.5–3 ม., พื้นพักระหว่างหลุมอย่างน้อย 5.5 ม., สิ่งกีดขวางวางกลางพื้นที่ยาว ≥ 9 ม., ก่อนประตูเปลี่ยนโซนเป็นพื้นเรียบเสมอ

### 3.5 โหมดจำลูกศร: สุ่มลำดับ + ตรวจคำตอบ — `Assets/Code/ArrowMemory/ArrowMemoryGame.cs`
**ใช้อธิบาย:** รอบแรกสุ่มลูกศร 2–4 ตัว (ตามระดับ) รอบต่อไปเพิ่มทีละ 1–2 ตัว ผู้เล่นเหยียบตามลำดับ ผิดเสียหัวใจ ผ่าน 3 รอบได้หัวใจ +1 และเวลา +30 วินาที

```csharp
// ★ เพิ่มลูกศรสุ่มต่อท้ายลำดับ (ลำดับยาวขึ้นทุกรอบ)
private void AppendDirections(int count)
{
    for (int i = 0; i < count; i++)
        sequence.Add(AllDirs[Random.Range(0, AllDirs.Length)]);
}

// ★ อ่านทิศจากแผ่นเหยียบ / WASD / ปุ่มลูกศร
private static Dir? ReadDirection()
{
    if (GameInputBindings.GetDown(GameAction.MoveUp)    || Input.GetKeyDown(KeyCode.W)) return Dir.Up;
    if (GameInputBindings.GetDown(GameAction.MoveDown)  || Input.GetKeyDown(KeyCode.S)) return Dir.Down;
    if (GameInputBindings.GetDown(GameAction.MoveLeft)  || Input.GetKeyDown(KeyCode.A)) return Dir.Left;
    if (GameInputBindings.GetDown(GameAction.MoveRight) || Input.GetKeyDown(KeyCode.D)) return Dir.Right;
    return null;
}
```
(ส่วนตรวจคำตอบใน `AcceptDirection`: ถูก = +คะแนน, ผิด = หัวใจ −1 แล้วแสดงลำดับให้ดูใหม่, ครบลำดับ = ผ่านรอบ)

### 3.6 การคำนวณแคลอรี่ (สูตร METs) — `Assets/Code/StageConfig.cs`
**ใช้อธิบาย:** โหมดถ้ำและโหมดวิ่งใช้สูตรมาตรฐาน ACSM: แคลอรี่ = (METs × 3.5 × น้ำหนักกก.) / 200 × นาที
(โหมดจำลูกศรคิดแบบประมาณจากจำนวนก้าว: จำนวนก้าว × `KcalPerStep`)

```csharp
// ★ คำนวณแคลอรี่ตามสูตร: (METs * 3.5 * น้ำหนัก) / 200 * เวลา (นาที)
public float CalculateCalories()
{
    float userWeight = PlayerPrefs.GetFloat("UserWeightKg", defaultWeightKg);
    float playTimeInMinutes = Time.timeSinceLevelLoad / 60f;
    return ((stageMETs * 3.5f * userWeight) / 200f) * playTimeInMinutes;
}
```
ค่า METs ที่ใช้: โหมดถ้ำค่าเริ่มต้น 4.0 (เคลื่อนไหวต่อเนื่อง ปรับได้ในแต่ละด่านที่ `StageConfig.stageMETs`), โหมดวิ่ง 5.0 (`CookieGameManager.exerciseMETs`)
ตัวอย่างคำนวณสำหรับรายงาน: น้ำหนัก 60 กก. เล่นโหมดวิ่ง 5 นาที → (5 × 3.5 × 60) / 200 × 5 = **26.25 kcal**

## 5. หัวข้อ "การออกแบบให้เหมาะกับแผ่นเหยียบ" (จุดเด่นของโปรเจกต์)

- ความเร็ววิ่งสูงสุดจำกัดที่ 1.4 เท่า และเว้นระยะสิ่งกีดขวาง เพราะผู้เล่นไม่มีราวจับ
- จำปุ่มกระโดดล่วงหน้า 0.3 วินาที (Jump Buffer) ให้เหยียบเร็วไปนิดก็ยังกระโดดได้
- หลุมกว้างไม่เกิน 3 ม. มีพื้นพักอย่างน้อย 5.5 ม. ก่อนประตูเปลี่ยนโซนเป็นพื้นเรียบ
- มอนสเตอร์หยุดเดินห่างสิ่งกีดขวางก่อนหน้าอย่างน้อย 8 ม. ให้มีเวลาเหยียบครั้งต่อไป
- บัฟ Blast หมดเวลาตอนอยู่เหนือหลุม จะลอยต่อจนถึงพื้นก่อน ไม่ตกหลุมแบบไม่ทันตั้งตัว
- โหมดจำลูกศรมีเสียงพูดบอกทิศ (บน/ล่าง/ซ้าย/ขวา) ช่วยผู้เล่นที่ไม่ได้มองจอตลอด
- ใช้ได้ทั้งแผ่นเหยียบ คีย์บอร์ด (WASD/ลูกศร) และเมาส์ ในทุกหน้าเมนู
- นับสถิติการเคลื่อนไหวจริง (ก้าว / ขุด / ปักธง / กระโดด / สไลด์) แสดงในหน้าสรุปผล
